const username = "iptv-paid-swf";
const repoName = "Skym3u.dev";
const folderName = "SkyM3U"; 

async function loadFilesFromGitHub() {
    const playlistContainer = document.getElementById("playlist");
    playlistContainer.innerHTML = `<p class="text-slate-400 text-center py-6 text-sm sm:text-base">Loading files...</p>`; 

    // GitHub API এর পরিবর্তে jsDelivr API বা Alternative endpoint ব্যবহার করে ফাইল লিস্ট ফেচ করা
    const apiUrl = `https://data.jsdelivr.com/v1/stats/packages/gh/${username}/${repoName}`;
    const contentsUrl = `https://api.github.com/repos/${username}/${repoName}/contents/${folderName}`;

    try {
        // প্রথমে সরাসরি GitHub Contents API দিয়ে চেষ্টার জন্য ফেস হেডার আপডেট করা
        let response = await fetch(contentsUrl, {
            headers: {
                'Accept': 'application/vnd.github.v3+json',
                'User-Agent': 'SkyM3U-Portal'
            }
        });
        
        if (!response.ok) {
            throw new Error("API failed");
        }

        let files = await response.json();
        playlistContainer.innerHTML = ""; 

        if (!Array.isArray(files) || files.length === 0) {
            playlistContainer.innerHTML = `<p class="text-slate-400 text-center py-6 text-sm sm:text-base">No files found in the folder.</p>`;
            return;
        }

        files.forEach(file => {
            if (file.type === "file") {
                let fileName = file.name; 
                let downloadUrl = `https://cdn.jsdelivr.net/gh/${username}/${repoName}@main/${folderName}/${fileName}`;

                let fileItem = `
                    <div class="bg-slate-700/40 hover:bg-slate-700/70 transition-all p-3 sm:p-4 rounded-xl flex flex-col sm:flex-row items-stretch sm:items-center justify-between gap-3 sm:gap-4 border border-slate-600/40 shadow-sm">
                        <div class="w-full sm:w-auto overflow-hidden">
                            <h3 class="font-medium text-slate-100 text-sm sm:text-base flex items-center gap-2 truncate">
                                <span class="text-emerald-400 shrink-0">📄</span> 
                                <span class="truncate" title="${fileName}">${fileName}</span>
                            </h3>
                        </div>
                        <button onclick="downloadOriginalFile('${downloadUrl}', '${fileName}')" class="w-full sm:w-auto bg-emerald-600 hover:bg-emerald-500 text-white font-medium px-4 sm:px-5 py-2.5 rounded-lg text-xs sm:text-sm text-center transition shadow-md flex items-center justify-center gap-2 cursor-pointer active:scale-95 shrink-0">
                            📥 Download
                        </button>
                    </div>
                `;
                playlistContainer.innerHTML += fileItem;
            }
        });

    } catch (error) {
        console.error("Fallback triggered:", error);
        
        // যদি গিটহাব এপিআই রেট লিমিটে পড়ে যায়, তবে ব্যাকআপ হিসেবে সরাসরি আপনার ফোল্ডারের চেনা ফাইলগুলো দিয়ে ফলব্যাক লিস্ট শো করবে যাতে সাইট ডাউন না থাকে
        loadFallbackFiles(playlistContainer);
    }
}

// ফোল্ডারের ব্যাকআপ লিস্ট যদি এপিআই কাজ না করে

    const fallbackList = [

        "1.m3u",
        "17-09-26_Rush_Sports.m3u"
    ];

    container.innerHTML = "";
    fallbackList.forEach(fileName => {
        let downloadUrl = `https://cdn.jsdelivr.net/gh/${username}/${repoName}@main/${folderName}/${fileName}`;
        let fileItem = `

                <div class="w-full sm:w-auto overflow-hidden">
                    <h3 class="font-medium text-slate-100 text-sm sm:text-base flex items-center gap-2 truncate">
                        <span class="text-emerald-400 shrink-0">📄</span> 
                        <span class="truncate" title="${fileName}">${fileName}</span>
                    </h3>
                </div>
                <button onclick="downloadOriginalFile('${downloadUrl}', '${fileName}')" class="w-full sm:w-auto bg-emerald-600 hover:bg-emerald-500 text-white font-medium px-4 sm:px-5 py-2.5 rounded-lg text-xs sm:text-sm text-center transition shadow-md flex items-center justify-center gap-2 cursor-pointer active:scale-95 shrink-0">
                    📥 Download
                </button>
            </div>
        `;
        container.innerHTML += fileItem;
    });
}

// ডাউনলোড ফাংশন
async function downloadOriginalFile(url, filename) {
    try {
        let response = await fetch(url);
        let blob = await response.blob();
        let blobUrl = window.URL.createObjectURL(blob);
        
        let a = document.createElement('a');
        a.href = blobUrl;
        a.download = filename;
        document.body.appendChild(a);
        a.click();
        
        document.body.removeChild(a);
        window.URL.revokeObjectURL(blobUrl);
    } catch (error) {
        console.error("Download error:", error);
        window.open(url, '_blank');
    }
}

loadFilesFromGitHub();
